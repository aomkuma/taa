from __future__ import annotations

import logging
from datetime import datetime, time, timedelta
from pathlib import Path
from zoneinfo import ZoneInfo

TH = ZoneInfo("Asia/Bangkok")
DAY_STARTS = time(7)  # the owner's trading day opens with the Asia window
DEFAULT_DIR = Path("data/plans")

log = logging.getLogger(__name__)


def trading_date(now: datetime) -> str:
    """The Thai trading day ``now`` belongs to, as ``YYYY-MM-DD``.

    02:00 Thai on Wednesday is still Tuesday's trading day; 07:00 Thai starts the new one.
    """
    th = now.astimezone(TH)
    return (th.date() - timedelta(days=1) if th.time() < DAY_STARTS else th.date()).isoformat()


class DayPlan:
    """Is today's plan written? Presence of the file is the whole answer."""

    def __init__(self, directory: Path | str = DEFAULT_DIR) -> None:
        self.directory = Path(directory)

    def path(self, now: datetime) -> Path:
        return self.directory / f"{trading_date(now)}.md"

    def exists(self, now: datetime) -> bool:
        """True when the day's plan file is there and not empty.

        An empty file is not a plan: a shell redirect that failed halfway would otherwise open the gate,
        and nor is one holding only whitespace (developer session, 2026-10-27).
        """
        p = self.path(now)
        try:
            return p.is_file() and bool(p.read_text(encoding="utf-8").strip())
        except OSError:  # pragma: no cover - a vanished mount is the same as no plan
            log.warning("cannot read the day plan at %s; treating the day as unplanned", p)
            return False

    def write(self, now: datetime, text: str) -> Path:
        """Record the day's plan. Refuses to overwrite one: a day has one plan, added to, not replaced."""
        p = self.path(now)
        if self.exists(now):
            raise FileExistsError(f"the plan for {trading_date(now)} is already written: {p}")
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_bytes(text.encode("utf-8"))
        log.info("the plan for %s is recorded (%s)", trading_date(now), p)
        return p

    BREAKERS = "BREAKERS"


@dataclass(frozen=True, slots=True)
class GateResult:
    mode: TradingMode
    failures: tuple[tuple[GateCondition, str], ...]

    @property
    def passed(self) -> bool:
        return not self.failures

    @property
    def reason(self) -> Reason | None:
        if self.passed:
            return None
        if any(cond is GateCondition.MODE for cond, _ in self.failures):
            return Reason.ORDERS_NOT_ALLOWED_IN_MODE
        return Reason.LIVE_GATE_FAILED

    @property
    def failed_conditions(self) -> set[GateCondition]:
        return {cond for cond, _ in self.failures}


def confirmation_phrase(login: int | None) -> str:
    return f"I-ACCEPT-LIVE-RISK-{login}"


def evaluate_gate(
    mode: TradingMode,
    env: EnvSettings,
    *,
    account: AccountSnapshot | None,
    terminal: TerminalSnapshot | None,
    risk_config_ok: bool,
    kill_switch_active: bool,
    latched_breakers: Sequence[str] = (),
    day_plan_ok: bool,  # P26: deliberately NO default - a forgotten argument must be a TypeError,
    day_plan_detail: str = "",  # not silent permission (developer session, 2026-10-27)
) -> GateResult:
    if not mode.may_send_broker_orders:
        return GateResult(mode, ((GateCondition.MODE, f"{mode.value} never sends broker orders"),))
    live = mode is TradingMode.LIVE
    failures: list[tuple[GateCondition, str]] = []

    def need(condition: GateCondition, ok: bool, detail: str) -> None:
        if not ok:
            failures.append((condition, detail))

    if env.TRADING_MODE is not mode:
        failures.append(
            (GateCondition.MODE, f"TRADING_MODE is {env.TRADING_MODE.value}, gate asked for {mode.value}")
        )
    if live:
        need(GateCondition.ENABLE_FLAG, env.ENABLE_LIVE_TRADING, "ENABLE_LIVE_TRADING is not true")
        need(
            GateCondition.CONFIRMATION,
            env.MT5_LOGIN is not None and confirmation_phrase(env.MT5_LOGIN) == env.LIVE_TRADING_CONFIRMATION,
            "LIVE_TRADING_CONFIRMATION does not match the phrase for MT5_LOGIN",
        )
    else:
        need(GateCondition.ENABLE_FLAG, env.ENABLE_DEMO_TRADING, "ENABLE_DEMO_TRADING is not true")
    need(GateCondition.DAY_PLAN, day_plan_ok, day_plan_detail or "the day's plan is not written (P26)")
    if account is None or terminal is None:
        failures.append((GateCondition.ACCOUNT_VERIFIED, "no account or terminal snapshot"))
    else:
        identity = verify_identity(env, account, terminal)
        need(GateCondition.ACCOUNT_VERIFIED, not identity, "; ".join(identity))
        expected = c.ACCOUNT_TRADE_MODE_REAL if live else c.ACCOUNT_TRADE_MODE_DEMO
        needed = c.ACCOUNT_TRADE_MODE_NAMES[expected]
        need(
            GateCondition.ACCOUNT_TRADE_MODE,
            account.trade_mode == expected,
            f"account is {account.trade_mode_name}, {mode.value} needs {needed}",
        )
        need(
            GateCondition.ACCOUNT_PERMISSIONS,
            account.trade_allowed and account.trade_expert,
            "account trade_allowed / trade_expert is off (investor password or broker restriction)",
        )
        need(
            GateCondition.TERMINAL_PERMISSIONS,
            terminal.trade_allowed and not terminal.tradeapi_disabled,
            "terminal Algo Trading is off or the Python API is disabled",
        )
    need(GateCondition.RISK_CONFIG, risk_config_ok, "risk configuration is invalid")
    need(GateCondition.KILL_SWITCH, not kill_switch_active, "kill switch is active")
    need(GateCondition.BREAKERS, not latched_breakers, f"latched breakers: {', '.join(latched_breakers)}")
    return GateResult(mode, tuple(failures))
"""Which side has the edge on a timeframe LARGER than the one the trade is entered on.

The owner's rule, 2026-10-27: *"every trade takes only the side with the edge on a timeframe larger than
the entry timeframe"*, and their correction the same day: *"D1 should not be that rigid - it is a
quantity, not a gate; the more momentum there is, the further that side runs."*

So the frame's lean is read as a **number**, not a label: the close's distance from its own 50-bar mean,
in that frame's own 14-bar ATR, signed (positive = leaning up). A trade is allowed when the lean is on
its side; the threshold is zero, not a fitted number.

**The frame is H4, measured** (`research/2026-10-26-three-bots/RULES_LOCKED.md`, rule set 2; 239 trades,
2023-09 .. 2026-10-24). Correlation of "lean behind the trade" with the trade's R: H4 +0.156, H1 +0.048,
D1 +0.030 - the middle frame carries it, nearer than the day and slower than the hour. Requiring the H4
lean to be on the trade's side moves the three bots from +0.102 to +0.210 R a trade, and the two halves
of the history agree to 0.007 R (+0.214 fitted, +0.207 never seen) - the reason this rule was taken and
an H1 one was not.
"""

from __future__ import annotations

import math
from collections.abc import Mapping

import pandas as pd

from app.core.enums import Side, Timeframe
from app.indicators.volatility import atr

MEAN_BARS = 50
ATR_BARS = 14
HOURS = {Timeframe.H1: 1, Timeframe.H4: 4, Timeframe.D1: 24}


def resample_from_h1(h1: pd.DataFrame, hours: int) -> pd.DataFrame:
    """Bars of ``hours`` from the lane's H1 frame, indexed by each bar's OPEN.

    App frames are stamped at the bar's CLOSE, so the index is shifted back one hour first; the research
    harness reads history stamped at the open and shifts nothing. Getting this backwards costs an hour of
    lookahead, which is how a day of conclusions went wrong on 2026-10-27.
    """
    opens = h1.copy()
    opens.index = pd.DatetimeIndex(opens.index) - pd.Timedelta(hours=1)
    rule = f"{hours}h" if hours < 24 else "1D"
    return opens.resample(rule).agg({"open": "first", "high": "max", "low": "min", "close": "last"}).dropna()


def edge_lean(
    frames: Mapping[Timeframe, pd.DataFrame],
    at: pd.Timestamp,
    entry: Timeframe,
    *,
    frame: Timeframe = Timeframe.H4,
) -> float | None:
    """The signed lean of ``frame``, or ``None`` when it cannot be read from a closed bar.

    ``None`` never means "allowed": a caller that cannot read the frame above its entry has no edge side
    and does not trade. ``entry`` must be smaller than ``frame`` - the rule is meaningless otherwise, and
    a ``ValueError`` here stops it being read from the entry frame by accident.
    """
    if entry.seconds >= frame.seconds:
        raise ValueError(f"the entry frame {entry.value} must be smaller than {frame.value}")
    hours = HOURS.get(frame)
    if hours is None:
        raise ValueError(f"{frame.value} is not read from the H1 frame")

    h1 = frames.get(Timeframe.H1)
    if h1 is None or "close" not in h1:
        return None
    bars = resample_from_h1(h1, hours)
    closed = bars[(bars.index + pd.Timedelta(hours=hours)) <= pd.Timestamp(at)]
    if len(closed) < MEAN_BARS + ATR_BARS:
        return None
    mean = float(closed.close.rolling(MEAN_BARS).mean().iloc[-1])
    width = float(atr(closed.high, closed.low, closed.close, ATR_BARS).iloc[-1])
    if not math.isfinite(mean) or not math.isfinite(width) or width <= 0:
        return None
    lean = (float(closed.close.iloc[-1]) - mean) / width
    # A non-finite lean must never reach a caller: `nan < 0` is False, so a NaN would sail through a
    # gate written as "refuse when the lean is negative" and open a trade on no evidence at all
    # (found by the developer session reviewing this gate while it was live, 2026-10-27).
    # It IS reachable, though I first wrote that it was not: a finite, positive but tiny ATR (1e-320)
    # overflows the division to infinity while passing both guards above. The developer session found
    # that and was right (2026-10-27); the case now has its own test.
    return lean if math.isfinite(lean) else None


def behind(lean: float | None, side: Side) -> float | None:
    """``lean`` seen from a trade's own side: positive means the frame leans the trade's way."""
    if lean is None:
        return None
    return lean if side is Side.BUY else -lean


def edge_side(
    frames: Mapping[Timeframe, pd.DataFrame],
    at: pd.Timestamp,
    entry: Timeframe,
    *,
    frame: Timeframe = Timeframe.H4,
) -> Side | None:
    """The side ``frame`` leans towards, for callers that want one answer rather than the number.

    A lean of exactly zero is on neither side and returns ``None``; the measured rule is "at least 0 on
    the trade's own side", so a strategy that cares about that boundary uses :func:`behind` instead.
    """
    lean = edge_lean(frames, at, entry, frame=frame)
    if lean is None or lean == 0:
        return None
    return Side.BUY if lean > 0 else Side.SELL

def restore_any() -> None:
    for f in sorted(BACKUP.glob("*.bak")) if BACKUP.exists() else []:
        head, body = f.read_text(encoding="utf-8").split("\n", 1)
        (ROOT / head).write_bytes(body.encode("utf-8"))
        print(f"restored {head} from an interrupted run")
        f.unlink()


def main() -> int:

    restore_any()
    if not run():
        print("the tests do not pass on the clean source - stopping")
        return 1
    missed = []
    for name, path, old, new in BUGS:
        f = ROOT / path
        src = f.read_bytes().decode("utf-8")
        if src.count(old) != 1 or old == new:
            print(f"SKIP    {name} (the text is not unique in {path})")
            missed.append(name + " [NOT PLANTED]")

        BACKUP.mkdir(exist_ok=True)
        parked = BACKUP / (path.replace("/", "_") + ".bak")
        parked.write_text(path + "\n" + src, encoding="utf-8")

        finally:

        if not caught:
            missed.append(name)
    print(f"\n{len(BUGS) - len(missed)}/{len(BUGS)} caught")
    for m in missed:
        print(f"  MISSED: {m}")

class BuyEveryBar(BaseStrategy):
    """Test strategy: a geometrically sane BUY on every bar (2 ATR stop, 4 ATR target)."""

    name = "buy_every_bar"

    def required_timeframes(self) -> tuple[Timeframe, ...]:
        return ()

    def warmup_bars(self) -> int:
        return 1

    def evaluate(self, ctx: StrategyContext) -> Signal:
        atr = ctx.market.atr
        if atr is None:
            return self.hold(ctx)
        ask = ctx.market.ask or ctx.market.entry.close
        return self.entry(
            ctx,
            Action.BUY,
            entry_price=ask,
            stop_loss=ask - 2 * atr,
            take_profit=ask + 4 * atr,
            conditions=[Condition("always", True)],
            score=50.0,
        )

**A â€” do not enter against the hour's own momentum.**

## Why set 1 was withdrawn, within two hours of writing it

Two independent faults, both found by the developer session reviewing my code:

1. **The wrong parameters.** Every measurement ran `bots.P()` â€” the dataclass defaults â€” instead of the

def at(hour: int, minute: int = 0, day: int = 27) -> datetime:
    """A moment given in Thai time, returned in UTC as the engine's clock hands it over."""
    return datetime(2026, 10, day, hour, minute, tzinfo=TH).astimezone(UTC)


class TestTheTradingDay:
    """07:00 Thai to 07:00 Thai, so a plan written on Tuesday morning still covers Wednesday's NY hours."""

    def test_the_day_turns_at_seven_in_the_morning_thai(self) -> None:
        assert trading_date(at(6, 59)) == "2026-10-26"  # still the previous trading day

def start_buyer(d: Demo) -> None:
    d.engine.start()
    strategy = BuyEveryBar()
    d.engine.strategies = StrategySet((strategy,))
    d.engine.magic = {strategy.name: MAGIC}


def run_until(d: Demo, event: EventType, limit: int = 80) -> None:
    for _ in range(limit):
        d.engine.cycle()
        if event in d.types():
            return
        d.clock.advance(30)
    raise AssertionError(f"{event} never happened; events: {d.types()}")

### `read_text()` behavior and missing-plan integration coverage

The whitespace check is content-correct for ordinary UTF-8 plans, but `read_text()` reads the **entire file into memory**. A very large plan can cause unnecessary memory use; a bounded read would be safer. Permission and other filesystem `OSError`s are caught and treated as unplanned. Invalid UTF-8 raises `UnicodeDecodeError`, which is **not** an `OSError` and is not caught in `DayPlan.exists()`. That exception does not yield permission, and the pre-send callback cannot reach `execution.send()` if it propagates, so the path appears fail-closed for orders; it does not return a normal `DAY_PLAN` refusal, and this case is untested. Treating decode errors as unplanned would make the contract explicit.

There is **no end-to-end DEMO engine test** that removes or omits the plan and proves no broker order is sent. ### `read_text()` behavior and missing-plan integration coverage

The whitespace check is content-correct for ordinary UTF-8 plans, but `read_text()` reads the **entire file into memory**. A very large plan can cause unnecessary memory use; a bounded read would be safer. Permission and other filesystem `OSError`s are caught and treated as unplanned. Invalid UTF-8 raises `UnicodeDecodeError`, which is **not** an `OSError` and is not caught in `DayPlan.exists()`. That exception does not yield permission, and the pre-send callback cannot reach `execution.send()` if it propagates, so the path appears fail-closed for orders; it does not return a normal `DAY_PLAN` refusal, and this case is untested. Treating decode errors as unplanned would make the contract explicit.

There is **no end-to-end DEMO engine test** that removes or omits the plan and proves no broker order is sent. The fixture writes a plan every time, and the current DEMO integration tests exercise success, kill-switch blocking, and flattening—not a missing-plan refusal. Unit tests cover `DayPlan.exists()` and the gate’s truth table, but not the full engine-to-order path for P26. This remains the key test hole.
