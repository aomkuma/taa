MEAN_BARS, ATR_BARS = 50, 14
FRAMES = {"H1": 1, "H4": 4, "D1": 24}  # hours a bar covers


def atr(frame: pd.DataFrame, n: int = ATR_BARS) -> pd.Series:
    """The app's own ATR, so research and production measure with one ruler.

    An earlier version used ``ewm(adjust=False)``, which seeds from the first true range, while the app's
    ``recursive_smooth`` seeds from the mean of the first full window - the two drift apart near the
    warmup (found by the developer session, 2026-10-27).
    """
    return app_atr(frame.high, frame.low, frame.close, n)


def momentum(h1: pd.DataFrame, hours: int) -> pd.DataFrame:
    """One row per CLOSED bar of the frame: its momentum, and the time it became readable.

    ``to_frame`` indexes a bar by its OPEN (see its docstring), so a bucket opening at ``t`` closes at
    ``t + hours`` and is readable only then. An earlier version shifted the index back an hour as if the
    frames were close-stamped - a double shift that made every bar readable an hour before it closed and
    put the H4 and D1 buckets an hour out (found by the developer session, 2026-10-27).
    """
    rule = f"{hours}h" if hours < 24 else "1D"
    f = h1.resample(rule).agg({"high": "max", "low": "min", "close": "last"}).dropna()
    out = pd.DataFrame(index=f.index)
    out["mom"] = (f.close - f.close.rolling(MEAN_BARS).mean()) / atr(f)
    out["readable"] = f.index + pd.Timedelta(hours=hours)
    return out.dropna()


def read_at(table: pd.DataFrame, at: pd.Timestamp) -> float:
    usable = table[table.readable <= at]
    return float(usable.mom.iloc[-1]) if len(usable) else float("nan")


def bins(df: pd.DataFrame, col: str, name: str) -> None:
    """Outcomes by how much momentum there was ON THE TRADE'S OWN SIDE."""
    x = df.dropna(subset=[col])
    if x.empty:
        print(f"\n{name}: no data")
        return
    # signed momentum turned into "momentum behind this trade": +ve means the frame leans our way
    behind = np.where(x.side == "BUY", x[col], -x[col])
    x = x.assign(behind=behind)
    edges = [-np.inf, -2, -1, 0, 1, 2, np.inf]
    labels = ["<-2", "-2..-1", "-1..0", "0..1", "1..2", ">2"]
    x = x.assign(band=pd.cut(x.behind, edges, labels=labels))
    print(f"\n### {name}\n")
    print("| momentum behind the trade | trades | win | R a trade | **median MFE (R)** | fit / unseen R |")
    print("|---|---|---|---|---|---|")
    for lab in labels:
        g = x[x.band == lab]
        if len(g) < 8:
            continue
        fit, uns = g[g.fitted], g[~g.fitted]
        both = (
            f"{fit.r.mean():+.2f} ({len(fit)}) / {uns.r.mean():+.2f} ({len(uns)})"
            if len(fit) and len(uns)
            else "-"
        )
        print(
            f"| {lab} | {len(g)} | {(g.r > 0).mean():.0%} | {g.r.mean():+.3f} | "
            f"**{g.mfe_r.median():.2f}** | {both} |"
        )
    c = x[["behind", "mfe_r"]].corr().iloc[0, 1]
    cr = x[["behind", "r"]].corr().iloc[0, 1]
    print(f"\ncorrelation with MFE **{c:+.3f}**, with the final R {cr:+.3f}  ({len(x)} trades)")

def day_end(at: pd.Timestamp) -> pd.Timestamp:
    """``run``'s own deadline for a trade opened at ``at``: 07:00 Bangkok the next morning."""
    th = at.tz_convert(bots.TH)
    return (pd.Timestamp(th.date(), tz=bots.TH) + pd.Timedelta(days=1, hours=7)).tz_convert("UTC")


def path_stats(m1: pd.DataFrame, t: bots.Trade, p: bots.P) -> dict:
    """Walk the real M1 path once and read off everything the entry's geometry needs.

    The counterfactual stops get the SAME DEADLINE the original trade had (07:00 Bangkok the next day,
    and ``max_hours``), not the moment the original happened to exit: the original exited early *because*
    it was stopped, so cutting the wider stop off there would answer the question with itself. The first
    version of this file did exactly that and gave a win rate that did not move with the stop width -
    which is impossible, and is what exposed the mistake.
    """
    x = m1[(m1.index >= t.at) & (m1.index <= t.extra["closed"])]
    if x.empty:
        return {}
    sell = t.side == "SELL"
    risk = abs(t.stop - t.entry)
    hi, lo = x.high.to_numpy(float), x.low.to_numpy(float)
    adverse = (hi.max() - t.entry) if sell else (t.entry - lo.min())
    favour = (t.entry - lo.min()) if sell else (hi.max() - t.entry)
    out = {"mae_r": adverse / risk, "mfe_r": favour / risk, "risk": risk}
    until = day_end(t.at)
    for k in WIDE:
        wider = bots.Trade(
            t.bot, t.side, t.at, t.entry, t.entry + k * risk if sell else t.entry - k * risk, t.tp
        )
        got = bots.settle(m1, wider, until, p)
        out[f"r_x{k}"] = got.r * k  # back into R of the ORIGINAL risk, so the k columns are comparable
        out[f"out_x{k}"] = got.out
    return out


def table(df: pd.DataFrame, title: str) -> None:

    print("| group | trades | median MAE (R) | 75th MAE | median MFE (R) | mean R |")
    print("|---|---|---|---|---|---|")
    for name, g in (("winners", win), ("losers", lose), ("all", df)):
        if g.empty:

        print(

            f"{g.mfe_r.median():.2f} | {g.r.mean():+.3f} |"
        )
    print("\n| stop width | mean R a trade | win rate | trades stopped out |")
    print("|---|---|---|---|")
    for k in WIDE:
        col = df[f"r_x{k}"]
        print(f"| x{k} | {col.mean():+.3f} | {(col > 0).mean():.0%} | {(col <= -0.9 * k).sum()} |")

from __future__ import annotations

import math
from collections.abc import Mapping
from dataclasses import dataclass
from typing import Literal

Wants = Literal["high", "low", "true", "false"]


@dataclass(frozen=True)
class Weight:
    """A reading and the nonnegative amount it contributes to a weighted decision."""

    name: str
    weight: float
    wants: Wants

    def __post_init__(self) -> None:
        if not self.name:
            raise ValueError("weight name must not be empty")
        if not math.isfinite(self.weight) or self.weight < 0:
            raise ValueError("weight must be finite and >= 0")
        if self.wants not in ("high", "low", "true", "false"):
            raise ValueError(f"unsupported wants value: {self.wants!r}")


@dataclass(frozen=True)
class Scale:
    """The readings and minimum normalized score required to qualify."""

    weights: tuple[Weight, ...]
    threshold: float

    def __post_init__(self) -> None:
        if not math.isfinite(self.threshold) or not 0 <= self.threshold <= 1:
            raise ValueError("threshold must be finite and in [0, 1]")
        names = [reading.name for reading in self.weights]
        if len(names) != len(set(names)):
            raise ValueError("weight names must be unique")


@dataclass(frozen=True)
class ScoreResult:
    """Normalized score, per-reading classifications, and threshold outcome."""

    score: float
    supported: list[str]
    opposed: list[str]
    missing: list[str]
    meets_threshold: bool


def score(readings: Mapping[str, float | bool | None], scale: Scale) -> ScoreResult:
    """Score available readings over their readable weight only.

    Numeric values must already be normalized to ``[0, 1]`` by the caller. ``high`` earns
    ``weight * value`` and ``low`` earns ``weight * (1 - value)``; a numeric reading supports when
    its contribution is at least half its weight, otherwise it opposes. Boolean readings earn full
    weight only when their value matches ``true`` or ``false``. Missing values are excluded from
    both numerator and denominator. The threshold is met at ``score >= threshold``; an empty or
    zero-readable-weight scale cannot meet it.
    """
    numerator = 0.0
    denominator = 0.0
    supported: list[str] = []
    opposed: list[str] = []
    missing: list[str] = []

    for item in scale.weights:
        value = readings.get(item.name)
        if value is None:
            missing.append(item.name)
            continue

        if isinstance(value, bool):
            if item.wants not in ("true", "false"):
                raise ValueError(f"{item.name}: boolean reading requires wants='true' or 'false'")
            agrees = value is (item.wants == "true")
            contribution = item.weight if agrees else 0.0
            if agrees:
                supported.append(item.name)
            else:
                opposed.append(item.name)
        else:
            if item.wants not in ("high", "low"):
                raise ValueError(f"{item.name}: numeric reading requires wants='high' or 'low'")
            numeric = float(value)
            if not math.isfinite(numeric) or not 0 <= numeric <= 1:
                raise ValueError(f"{item.name}: numeric reading must be finite and in [0, 1]")
            fraction = numeric if item.wants == "high" else 1.0 - numeric
            contribution = item.weight * fraction
            (supported if fraction >= 0.5 else opposed).append(item.name)
        numerator += contribution
        denominator += item.weight

    total = numerator / denominator if denominator > 0 else 0.0
    meets = denominator > 0 and total >= scale.threshold
    return ScoreResult(total, supported, opposed, missing, meets)

def stepping(sws: list) -> str:
    """'DOWN' (lower high and lower low), 'UP', or 'FLAT' - from the last two confirmed swings of each kind."""
    highs = [s for s in sws if s.kind == 1]
    lows = [s for s in sws if s.kind == -1]
    if len(highs) < 2 or len(lows) < 2:
        return "FLAT"
    if highs[-1].price < highs[-2].price and lows[-1].price < lows[-2].price:
        return "DOWN"
    if highs[-1].price > highs[-2].price and lows[-1].price > lows[-2].price:
        return "UP"
    return "FLAT"

+"""Measure decision-time readings against shipped-bot trade outcomes.
+
+Run from the repository root with ``.venv\\Scripts\\python.exe
+research/2026-10-26-three-bots/weights.py``. This reads the frozen trade CSV and
+prints summaries; it does not regenerate the trade history or write any files.
+
+Only columns known at entry are treated as candidate readings. ``r``, exit
+labels, excursions, and counterfactual stops are outcomes and never predictors.
+The momentum columns are changed to "behind the trade" by multiplying the
+stored signed value by +1 for BUY and -1 for SELL; side is known at entry.
+The momentum producer's read_at() accepts a frame bar only when its readable
+time (bar open + frame duration) is <= the decision time. In particular, for
+a trade at 2026-10-27 10:00 UTC, an H4 bar opened at 08:00 closes at 12:00 and
+is not readable; the preceding H4 bar opened at 04:00 closes at 08:00 and is
+readable. The history index is the bar OPEN. The app frames are close-stamped;
+momentum.py explicitly uses the history convention and does not shift the
+decision timestamp. This hand-worked boundary case is the point-in-time check.
+
+No indicators beyond the CSV are introduced here, so there are no additional
+bar joins or timing assumptions.
+"""
+
+from __future__ import annotations
+
+from pathlib import Path
+
+import numpy as np
+import pandas as pd
+
+HERE = Path(__file__).resolve().parent
+CSV = HERE / "momentum_trades.csv"
+NUMERIC = ("mom_H1", "mom_H4", "mom_D1", "risk")
+CATEGORICAL = ("bot", "side", "regime", "window")
+SEED = 20261027
+PERMUTATIONS = 5000
+
+
+def auc_higher_is_better(values: np.ndarray, wins: np.ndarray) -> float:
+    """Pairwise win/loss AUC; 0.5 is chance, >0.5 favors larger readings."""
+    positive, negative = values[wins], values[~wins]
+    if not len(positive) or not len(negative):
+        return float("nan")
+    pair = positive[:, None] - negative[None, :]
+    return float(((pair > 0).sum() + 0.5 * (pair == 0).sum()) / pair.size)
+
+
+def halves(df: pd.DataFrame):
+    yield "fit", df[df.fitted]
+    yield "unseen", df[~df.fitted]
+
+
+def numeric_summary(df: pd.DataFrame, col: str) -> dict:
+    x = df[[col, "r", "win"]].dropna()
+    med = x[col].median()
+    high, low = x[x[col] > med], x[x[col] <= med]
+    positive, negative = x[x[col] >= 0], x[x[col] < 0]
+    return {
+        "n": len(x),
+        "wins": int(x.win.sum()),
+        "auc": auc_higher_is_better(x[col].to_numpy(), x.win.to_numpy(bool)),
+        "corr_r": x[col].corr(x.r),
+        "median": med,
+        "hi_n": len(high),
+        "hi_win": high.win.mean(),
+        "hi_r": high.r.mean(),
+        "lo_n": len(low),
+        "lo_win": low.win.mean(),
+        "lo_r": low.r.mean(),
+        "median_win_delta": high.win.mean() - low.win.mean(),
+        "median_r_delta": high.r.mean() - low.r.mean(),
+        "pos_n": len(positive),
+        "pos_win": positive.win.mean(),
+        "pos_r": positive.r.mean(),
+        "neg_n": len(negative),
+        "neg_win": negative.win.mean(),
+        "neg_r": negative.r.mean(),
+        "zero_r_delta": positive.r.mean() - negative.r.mean(),
+    }
+
+
+def categorical_summary(df: pd.DataFrame, col: str, level: str) -> dict:
+    inside = df[col].eq(level)
+    a, b = df[inside], df[~inside]
+    return {
+        "level_n": len(a),
+        "other_n": len(b),
+        "level_win": a.win.mean() if len(a) else float("nan"),
+        "other_win": b.win.mean() if len(b) else float("nan"),
+        "win_delta": a.win.mean() - b.win.mean() if len(a) and len(b) else float("nan"),
+        "level_r": a.r.mean() if len(a) else float("nan"),
+        "other_r": b.r.mean() if len(b) else float("nan"),
+        "r_delta": a.r.mean() - b.r.mean() if len(a) and len(b) else float("nan"),
+    }
+
+
+def candidate_matrix(df: pd.DataFrame) -> pd.DataFrame:
+    """16 predeclared linear candidates for the best-of-N null calculation."""
+    out = df[list(NUMERIC)].copy()
+    for col in NUMERIC:
+        out[col] = out[col].fillna(out[col].median())
+    for col in CATEGORICAL:
+        dummies = pd.get_dummies(df[col], prefix=col, dtype=float)
+        out = pd.concat([out, dummies], axis=1)
+    return out
+
+
+def best_of_n_null(df: pd.DataFrame) -> tuple[int, float, float]:
+    """Permutation null for max absolute feature/R correlation over 16 candidates."""
+    x = candidate_matrix(df).to_numpy(float)
+    y = df.r.to_numpy(float)
+    x = (x - x.mean(axis=0)) / x.std(axis=0)
+    x = x[:, np.isfinite(x.std(axis=0))] if False else x
+    # Candidate columns are nonconstant in this dataset. Centering y after each
+    # shuffle preserves its observed distribution while breaking all alignment.
+    x = x[:, np.std(x, axis=0) > 0]
+    x = x / np.sqrt((x * x).sum(axis=0))
+    y = y - y.mean()
+    observed = float(np.max(np.abs(x.T @ y) / np.sqrt((y * y).sum())))
+    rng = np.random.default_rng(SEED)
+    maxima = np.empty(PERMUTATIONS)
+    for i in range(PERMUTATIONS):
+        yp = rng.permutation(y)
+        maxima[i] = np.max(np.abs(x.T @ yp) / np.sqrt((yp * yp).sum()))
+    return x.shape[1], observed, float(np.quantile(maxima, 0.95))
+

+def fmt(value: float, digits: int = 3) -> str:
+    return "—" if pd.isna(value) else f"{value:+.{digits}f}"
+

+def main() -> None:
+    d = pd.read_csv(CSV)
+    d["at"] = pd.to_datetime(d["at"], utc=True)
+    d["fitted"] = d["fitted"].astype(bool)
+    d["mom_D1"] = pd.to_numeric(d["mom_D1"], errors="coerce")
+    d["win"] = d.r > 0
+    for col in ("mom_H1", "mom_H4", "mom_D1"):
+        d[col] = d[col] * np.where(d.side.eq("BUY"), 1.0, -1.0)
+
+    print(f"Trades: {len(d)}; fit={int(d.fitted.sum())}; unseen={int((~d.fitted).sum())}")
+    print("\nNUMERIC (AUC: larger reading ranks winners; median/zero splits are descriptive)")
+    for col in NUMERIC:

+        for name, frame in halves(d):
+            s = numeric_summary(frame, col)
+            print(
+                f" {name:6s} n={s['n']:3d} wins={s['wins']:3d} auc={s['auc']:.3f} "
+                f"corrR={fmt(s['corr_r'])} med={s['median']:.3f}; "
+                f"above-median {s['hi_n']} win={s['hi_win']:.3f} R={fmt(s['hi_r'])}, "
+                f"at/below {s['lo_n']} win={s['lo_win']:.3f} R={fmt(s['lo_r'])}; "
+                f"deltaR={fmt(s['median_r_delta'])}; "
+                f">=0 {s['pos_n']} R={fmt(s['pos_r'])}, <0 {s['neg_n']} R={fmt(s['neg_r'])}, "
+                f"deltaR={fmt(s['zero_r_delta'])}"
+            )
+
+    print("\nCATEGORICAL (each level versus all other levels; winDelta and RDelta)")
+    for col in CATEGORICAL:
+        print(f"\n{col}")
+        for level in sorted(d[col].dropna().unique()):
+            cells = []
+            for name, frame in halves(d):
+                s = categorical_summary(frame, col, level)
+                cells.append(
+                    f"{name} {s['level_n']}/{s['other_n']} "
+                    f"win={fmt(s['level_win'])}/{fmt(s['other_win'])} "
+                    f"dWin={fmt(s['win_delta'])} R={fmt(s['level_r'])}/{fmt(s['other_r'])} "
+                    f"dR={fmt(s['r_delta'])}"
+                )
+            print(f" {level}: " + " | ".join(cells))
+
+    print("\nMOMENTUM OVERLAP: Pearson correlations of behind-trade values")
+    for name, frame in halves(d):
+        print(f"\n{name}\n{frame[['mom_H1', 'mom_H4', 'mom_D1']].corr().round(3)}")
+    print("\nBOT x SIDE counts")
+    print(pd.crosstab(d.bot, d.side).to_string())
+    print("\nBOT x REGIME counts")
+    print(pd.crosstab(d.bot, d.regime).to_string())
+    k, obs, null95 = best_of_n_null(d)
+    print(
+        f"\nBest-of-N null: {k} predeclared numeric/one-hot columns; observed max |corr(R)|="
+        f"{obs:.3f}; 95th percentile after {PERMUTATIONS} R permutations={null95:.3f}"
+    )
+    print("\nHALF BASELINES")
+    for name, frame in halves(d):
+        print(f"{name}: n={len(frame)} win={frame.win.mean():.3f} meanR={frame.r.mean():+.3f}")
+    print(f"D1 missing: {int(d.mom_D1.isna().sum())} trades")
+
+
+if __name__ == "__main__":
+    main()

+
+Run from the repository root with ``.venv\\Scripts\\python.exe
+research/2026-10-26-three-bots/weights.py``. This reads the frozen trade CSV and

+

+labels, excursions, and counterfactual stops are outcomes and never predictors.
+The momentum columns are changed to "behind the trade" by multiplying the
+stored signed value by +1 for BUY and -1 for SELL; side is known at entry.
+The momentum producer's read_at() accepts a frame bar only when its readable
+time (bar open + frame duration) is <= the decision time. In particular, for
+a trade at 2026-10-27 10:00 UTC, an H4 bar opened at 08:00 closes at 12:00 and
+is not readable; the preceding H4 bar opened at 04:00 closes at 08:00 and is
+readable. The history index is the bar OPEN. The app frames are close-stamped;
+momentum.py explicitly uses the history convention and does not shift the
+decision timestamp. This hand-worked boundary case is the point-in-time check.
+
+No indicators beyond the CSV are introduced here, so there are no additional
+bar joins or timing assumptions.

+
+from __future__ import annotations
+
+from pathlib import Path
+
+import numpy as np
+import pandas as pd
+
+HERE = Path(__file__).resolve().parent
+CSV = HERE / "momentum_trades.csv"
+NUMERIC = ("mom_H1", "mom_H4", "mom_D1", "risk")
+CATEGORICAL = ("bot", "side", "regime", "window")
+SEED = 20261027
+PERMUTATIONS = 5000
+
+
+def auc_higher_is_better(values: np.ndarray, wins: np.ndarray) -> float:
+    """Pairwise win/loss AUC; 0.5 is chance, >0.5 favors larger readings."""
+    positive, negative = values[wins], values[~wins]
+    if not len(positive) or not len(negative):
+        return float("nan")
+    pair = positive[:, None] - negative[None, :]
+    return float(((pair > 0).sum() + 0.5 * (pair == 0).sum()) / pair.size)
+
+
+def halves(df: pd.DataFrame):
+    yield "fit", df[df.fitted]
+    yield "unseen", df[~df.fitted]
+
+
+def numeric_summary(df: pd.DataFrame, col: str) -> dict:
+    x = df[[col, "r", "win"]].dropna()
+    med = x[col].median()
+    high, low = x[x[col] > med], x[x[col] <= med]
+    positive, negative = x[x[col] >= 0], x[x[col] < 0]
+    return {
+        "n": len(x),
+        "wins": int(x.win.sum()),
+        "auc": auc_higher_is_better(x[col].to_numpy(), x.win.to_numpy(bool)),
+        "corr_r": x[col].corr(x.r),
+        "median": med,
+        "hi_n": len(high),
+        "hi_win": high.win.mean(),
+        "hi_r": high.r.mean(),
+        "lo_n": len(low),
+        "lo_win": low.win.mean(),
+        "lo_r": low.r.mean(),
+        "median_win_delta": high.win.mean() - low.win.mean(),
+        "median_r_delta": high.r.mean() - low.r.mean(),
+        "pos_n": len(positive),
+        "pos_win": positive.win.mean(),
+        "pos_r": positive.r.mean(),
+        "neg_n": len(negative),
+        "neg_win": negative.win.mean(),
+        "neg_r": negative.r.mean(),
+        "zero_r_delta": positive.r.mean() - negative.r.mean(),
+    }
+
+
+def categorical_summary(df: pd.DataFrame, col: str, level: str) -> dict:
+    inside = df[col].eq(level)
+    a, b = df[inside], df[~inside]
+    return {
+        "level_n": len(a),
+        "other_n": len(b),
+        "level_win": a.win.mean() if len(a) else float("nan"),
+        "other_win": b.win.mean() if len(b) else float("nan"),
+        "win_delta": a.win.mean() - b.win.mean() if len(a) and len(b) else float("nan"),
+        "level_r": a.r.mean() if len(a) else float("nan"),
+        "other_r": b.r.mean() if len(b) else float("nan"),
+        "r_delta": a.r.mean() - b.r.mean() if len(a) and len(b) else float("nan"),
+    }
+
+
+def candidate_matrix(df: pd.DataFrame) -> pd.DataFrame:
+    """16 predeclared linear candidates for the best-of-N null calculation."""
+    out = df[list(NUMERIC)].copy()
+    for col in NUMERIC:
+        out[col] = out[col].fillna(out[col].median())
+    for col in CATEGORICAL:
+        dummies = pd.get_dummies(df[col], prefix=col, dtype=float)
+        out = pd.concat([out, dummies], axis=1)
+    return out
+
+
+def best_of_n_null(df: pd.DataFrame) -> tuple[int, float, float]:
+    """Permutation null for max absolute feature/R correlation over 16 candidates."""
+    x = candidate_matrix(df).to_numpy(float)
+    y = df.r.to_numpy(float)
+    x = (x - x.mean(axis=0)) / x.std(axis=0)
+    # Candidate columns are nonconstant in this dataset. Centering y after each
+    # shuffle preserves its observed distribution while breaking all alignment.
+    x = x[:, np.std(x, axis=0) > 0]
+    x = x / np.sqrt((x * x).sum(axis=0))
+    y = y - y.mean()
+    observed = float(np.max(np.abs(x.T @ y) / np.sqrt((y * y).sum())))
+    rng = np.random.default_rng(SEED)
+    maxima = np.empty(PERMUTATIONS)
+    for i in range(PERMUTATIONS):
+        yp = rng.permutation(y)
+        maxima[i] = np.max(np.abs(x.T @ yp) / np.sqrt((yp * yp).sum()))
+    return x.shape[1], observed, float(np.quantile(maxima, 0.95))
+
+
+def fmt(value: float, digits: int = 3) -> str:
+    return "—" if pd.isna(value) else f"{value:+.{digits}f}"
+

+def main() -> None:
+    d = pd.read_csv(CSV)
+    d["at"] = pd.to_datetime(d["at"], utc=True)
+    d["fitted"] = d["fitted"].astype(bool)
+    d["mom_D1"] = pd.to_numeric(d["mom_D1"], errors="coerce")
+    d["win"] = d.r > 0

+                f"corrR={fmt(s['corr_r'])} med={s['median']:.3f}; "
+                f"above-median {s['hi_n']} win={s['hi_win']:.3f} R={fmt(s['hi_r'])}, "

+                f"deltaR={fmt(s['median_r_delta'])}; "

+    for col in CATEGORICAL:
+        print(f"\n{col}")
+        for level in sorted(d[col].dropna().unique()):
+            cells = []
+            for name, frame in halves(d):
+                s = categorical_summary(frame, col, level)
+                cells.append(
+                    f"{name} {s['level_n']}/{s['other_n']} "
+                    f"win={fmt(s['level_win'])}/{fmt(s['other_win'])} "
+                    f"dWin={fmt(s['win_delta'])} R={fmt(s['level_r'])}/{fmt(s['other_r'])} "
+                    f"dR={fmt(s['r_delta'])}"

+            print(f" {level}: " + " | ".join(cells))
+
+    print("\nMOMENTUM OVERLAP: Pearson correlations of behind-trade values")
+    for name, frame in halves(d):
+        print(f"\n{name}\n{frame[['mom_H1', 'mom_H4', 'mom_D1']].corr().round(3)}")

+    print(
+        f"\nBest-of-N null: {k} predeclared numeric/one-hot columns; observed max |corr(R)|="
+        f"{obs:.3f}; 95th percentile after {PERMUTATIONS} R permutations={null95:.3f}"
+    )
+    print("\nHALF BASELINES")
+    for name, frame in halves(d):
+        print(f"{name}: n={len(frame)} win={frame.win.mean():.3f} meanR={frame.r.mean():+.3f}")
+    print(f"D1 missing: {int(d.mom_D1.isna().sum())} trades")
+
+
+if __name__ == "__main__":
+    main()

+# Reading weights: evidence, not yet a usable scale
+
+**Result:** I do not recommend assigning a positive trading weight to any reading from this sample. H4 momentum is the strongest lead, but the required direction is not stable across the two supplied halves. The proposed scale therefore has zero weight on every tested reading until forward evidence resolves the split. This is a measurement result, not a change to the frozen rule set 2.
+
+Run the reproducible analysis with:
+
+```powershell

+```

+The script reads `momentum_trades.csv`, prints the statistics below, and writes no files. It does not regenerate trades. “Fit” means the saved `fitted=True` period (n=95); “never seen” means `False` (n=144). They are the requested partitions, not a clean future test: the never-seen rows are older than the fit window. The all-trade baseline is +0.226 R / 40.0% wins in fit and +0.020 R / 35.4% wins in never-seen.
+
+## Timing and eligible readings
+
+I used only decision-time CSV columns: `bot`, `side`, `window`, `regime`, `risk`, and `mom_H1`, `mom_H4`, `mom_D1`. Outcomes (`r`, `out`, MAE/MFE and counterfactual stops) were not predictors. “Momentum behind” is the stored signed momentum multiplied by +1 for BUY and -1 for SELL; the side is already known at the entry decision. `risk` is the entry stop distance.
+
+The producer `momentum.py` forms each frame from history whose index is the bar **open**, then sets `readable = open + frame duration`; `read_at` takes only `readable <= at`. The app live frame is close-stamped, so applying an extra index shift would be wrong. Hand check at a decision `at=2026-10-27 10:00 UTC`: the H4 bucket opened at 08:00 closes at 12:00 and must be excluded; the prior H4 bucket opened at 04:00 closes at 08:00 and may be used. This is the history convention actually used by the CSV producer. No extra indicator or bar join was added here.

+## Numeric readings
+

It is descriptive, not a fitted threshold. “Zero ΔR” is the mean R for momentum ≥0 minus the mean R below 0; for risk, zero is not a useful split and is omitted. AUC and mean R answer different questions.

+## Overlap and selection premium
+
+Momentum readings overlap. Correlations of momentum-behind values are H1/H4 **0.327 fit, 0.267 unseen**; H4/D1 **0.618, 0.411**; H1/D1 **0.216, −0.071**. H4 and D1 especially can double-count similar directional context. The categorical readings overlap too: all 84 DOWN trades are SELL and all 77 UP trades are BUY; only RANGE has both sides (40 BUY, 38 SELL). Bot, side, and regime should not be assigned independent full-strength weights from these marginal tables.
+
+The script estimates best-of-N optimism with 5,000 permutations of R against the same 16 predeclared numeric/one-hot columns (four numeric columns plus categorical levels). The observed maximum absolute feature/R correlation is **0.156**; the 95th percentile of the null maximum is **0.187**. Thus a best-looking marginal correlation this large is not surprising under a no-association model after searching those 16 columns. This estimate is only for linear correlation in this small sample; it does not account for all possible cutoffs or analyst choices, so it is not a correction that makes a candidate significant.

+## Proposed weights and next measurement
+
+For the inert `Weight` / `Scale` vessel, the proposed evidence-based weights from this study are:
+
+| Reading | Weight | Direction | Why |
+|---|---:|---|---|
+| H1 momentum behind | 0.00 | high | Win and mean-R effects reverse between halves. |
+| H4 momentum behind | 0.00 | high | Best lead, but both win ranking and mean-R contrast fail the stable-sign requirement. Keep measuring; do not encode the lock’s subgroup mean as a weight. |
+| D1 momentum behind | 0.00 | high | AUC is chance in both halves; median-split R effect reverses. |
+| Risk / stop distance | 0.00 | — | Win ranking does not translate to R, especially outside fit. |
+| Bot, side, regime, session window | 0.00 | — | Marginal effects flip, overlap structurally, or rely on thin groups. |

+This is an explicit **no-positive-weight** proposal, rather than a qualified production score: the current readings do not pass the locked sign rule. No threshold is proposed. Rule set 2 remains a separately frozen historical rule and should be judged on its forward journal; this file does not revise it.
+
+**Would not take:** H1, D1, risk, bot, regime, session-window, or side weights from these marginal results. I would also not count H4 and D1 as independent evidence if later data supports both; they correlate substantially.
+
+**Measure next:** prospectively log every eligible entry decision from 2026-10-28 onward with its point-in-time readings, including H4 values for entries accepted and refused by the locked H4 ≥0 rule, then compare realized mean R and win separation on untouched forward data. That supplies the missing comparator and avoids selecting only the trades the current rule allowed.

+        "median": med,
+        "hi_n": len(high),
+        "hi_win": high.win.mean(),
+        "hi_r": high.r.mean(),
+        "lo_n": len(low),
+        "lo_win": low.win.mean(),
+        "lo_r": low.r.mean(),
+        "median_win_delta": high.win.mean() - low.win.mean(),
+        "median_r_delta": high.r.mean() - low.r.mean(),
+        "pos_n": len(positive),
+        "pos_win": positive.win.mean(),
+        "pos_r": positive.r.mean(),
+        "neg_n": len(negative),
+        "neg_win": negative.win.mean(),
+        "neg_r": negative.r.mean(),
+        "zero_r_delta": positive.r.mean() - negative.r.mean(),
+    }
+
+
+def categorical_summary(df: pd.DataFrame, col: str, level: str) -> dict:
+    inside = df[col].eq(level)
+    a, b = df[inside], df[~inside]
+    return {
+        "level_n": len(a),
+        "other_n": len(b),
+        "level_win": a.win.mean() if len(a) else float("nan"),
+        "other_win": b.win.mean() if len(b) else float("nan"),
+        "win_delta": a.win.mean() - b.win.mean() if len(a) and len(b) else float("nan"),
+        "level_r": a.r.mean() if len(a) else float("nan"),
+        "other_r": b.r.mean() if len(b) else float("nan"),
+        "r_delta": a.r.mean() - b.r.mean() if len(a) and len(b) else float("nan"),
+    }
+
+
+def candidate_matrix(df: pd.DataFrame) -> pd.DataFrame:
+    """16 predeclared linear candidates for the best-of-N null calculation."""
+    out = df[list(NUMERIC)].copy()
+    for col in NUMERIC:
+        out[col] = out[col].fillna(out[col].median())
+    for col in CATEGORICAL:
+        dummies = pd.get_dummies(df[col], prefix=col, dtype=float)
+        out = pd.concat([out, dummies], axis=1)
+    return out
+
+
+def best_of_n_null(df: pd.DataFrame) -> tuple[int, float, float]:
+    """Permutation null for max absolute feature/R correlation over 16 candidates."""
+    x = candidate_matrix(df).to_numpy(float)
+    y = df.r.to_numpy(float)
+    x = (x - x.mean(axis=0)) / x.std(axis=0)
+    # Candidate columns are nonconstant in this dataset. Centering y after each
+    # shuffle preserves its observed distribution while breaking all alignment.
+    x = x[:, np.std(x, axis=0) > 0]
+    x = x / np.sqrt((x * x).sum(axis=0))
+    y = y - y.mean()
+    observed = float(np.max(np.abs(x.T @ y) / np.sqrt((y * y).sum())))
+    rng = np.random.default_rng(SEED)
+    maxima = np.empty(PERMUTATIONS)
+    for i in range(PERMUTATIONS):
+        yp = rng.permutation(y)
+        maxima[i] = np.max(np.abs(x.T @ yp) / np.sqrt((yp * yp).sum()))
+    return x.shape[1], observed, float(np.quantile(maxima, 0.95))
+
+
+def fmt(value: float, digits: int = 3) -> str:
+    return "—" if pd.isna(value) else f"{value:+.{digits}f}"
+

+                    f"dR={fmt(s['r_delta'])}"
+                )

+

+    for name, frame in halves(d):
+        print(f"\n{name}\n{frame[['mom_H1', 'mom_H4', 'mom_D1']].corr().round(3)}")
+    print("\nBOT x SIDE counts")

+    print("\nBOT x REGIME counts")
+    print(pd.crosstab(d.bot, d.regime).to_string())
+    k, obs, null95 = best_of_n_null(d)
