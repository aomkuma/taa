from __future__ import annotations

from collections.abc import Mapping

import pandas as pd

from app.core.enums import Side, Timeframe
from app.indicators.volatility import atr

MEAN_BARS = 50
ATR_BARS = 14
HOURS = {Timeframe.H1: 1, Timeframe.H4: 4, Timeframe.D1: 24}


def resample_from_h1(h1: pd.DataFrame, hours: int) -> pd.DataFrame:
    """Bars of ``hours`` from the lane's H1 frame, indexed by each bar's OPEN.

    App frames are stamped at the bar's CLOSE, so the index is shifted back one hour first; the research
    harness reads history stamped at the open and shifts nothing. Getting this backwards costs an hour of
    lookahead, which is how a day of conclusions went wrong on 2026-10-27.
    """
    opens = h1.copy()
    opens.index = pd.DatetimeIndex(opens.index) - pd.Timedelta(hours=1)
    rule = f"{hours}h" if hours < 24 else "1D"
    return opens.resample(rule).agg({"open": "first", "high": "max", "low": "min", "close": "last"}).dropna()


def edge_lean(
    frames: Mapping[Timeframe, pd.DataFrame],
    at: pd.Timestamp,
    entry: Timeframe,
    *,
    frame: Timeframe = Timeframe.H4,
) -> float | None:
    """The signed lean of ``frame``, or ``None`` when it cannot be read from a closed bar.

    ``None`` never means "allowed": a caller that cannot read the frame above its entry has no edge side
    and does not trade. ``entry`` must be smaller than ``frame`` - the rule is meaningless otherwise, and
    a ``ValueError`` here stops it being read from the entry frame by accident.
    """
    if entry.seconds >= frame.seconds:
        raise ValueError(f"the entry frame {entry.value} must be smaller than {frame.value}")
    hours = HOURS.get(frame)
    if hours is None:
        raise ValueError(f"{frame.value} is not read from the H1 frame")

    h1 = frames.get(Timeframe.H1)
    if h1 is None or "close" not in h1:
        return None
    bars = resample_from_h1(h1, hours)
    closed = bars[(bars.index + pd.Timedelta(hours=hours)) <= pd.Timestamp(at)]
    if len(closed) < MEAN_BARS + ATR_BARS:
        return None
    mean = float(closed.close.rolling(MEAN_BARS).mean().iloc[-1])
    width = float(atr(closed.high, closed.low, closed.close, ATR_BARS).iloc[-1])
    if not width > 0 or pd.isna(mean):
        return None
    return (float(closed.close.iloc[-1]) - mean) / width


def behind(lean: float | None, side: Side) -> float | None:
    """``lean`` seen from a trade's own side: positive means the frame leans the trade's way."""
    if lean is None:
        return None
    return lean if side is Side.BUY else -lean


def edge_side(
    frames: Mapping[Timeframe, pd.DataFrame],
    at: pd.Timestamp,
    entry: Timeframe,
    *,
    frame: Timeframe = Timeframe.H4,
) -> Side | None:
    """The side ``frame`` leans towards, for callers that want one answer rather than the number.

    A lean of exactly zero is on neither side and returns ``None``; the measured rule is "at least 0 on
    the trade's own side", so a strategy that cares about that boundary uses :func:`behind` instead.
    """
    lean = edge_lean(frames, at, entry, frame=frame)
    if lean is None or lean == 0:
        return None
    return Side.BUY if lean > 0 else Side.SELL

def in_window(at: pd.Timestamp) -> bool:
    """Research windows use a start-exclusive, end-inclusive close timestamp."""
    utc = at.tz_convert("UTC")
    bangkok = utc.tz_convert(BANGKOK)
    new_york = utc.tz_convert(NEW_YORK)
    bt, nt = bangkok.time(), new_york.time()
    return (time(7) < bt <= time(9)) or (time(13) < bt <= time(15)) or (time(8) < nt <= time(11, 30))


def stepped(swings: list[Swing]) -> str:
    highs = [s.price for s in swings if s.kind == 1]
    lows = [s.price for s in swings if s.kind == -1]
    if len(highs) < 2 or len(lows) < 2:
        return "FLAT"
    if highs[-1] < highs[-2] and lows[-1] < lows[-2]:
        return "DOWN"
    if highs[-1] > highs[-2] and lows[-1] > lows[-2]:
        return "UP"
    return "FLAT"


def running_push_exceeded(run: float, risk: float) -> bool:
    """Apply the measured strict 1.2 R cutoff to the preceding M1 run."""
    return run > 1.2 * risk


def break_even_advice(ctx: StrategyContext, pos: OpenPosition) -> PositionAdvice | None:
    """Replay closed M1 bars so a stop crossed before a threshold touch keeps precedence."""
    if (pos.side is Side.BUY and pos.sl >= pos.entry) or (pos.side is Side.SELL and pos.sl <= pos.entry):
        return None
    m1 = ctx.frame(Timeframe.M1)
    bars = m1[(m1.index >= pd.Timestamp(pos.entry_time)) & (m1.index <= pd.Timestamp(ctx.decision_time_utc))]
    risk = abs(pos.initial_sl - pos.entry)
    if risk <= 0 or bars.empty:
        return None
    sell = pos.side is Side.SELL
    stop = pos.sl
    moved = False
    highs = bars["high"].to_numpy(dtype=float)
    lows = bars["low"].to_numpy(dtype=float)
    for high, low in zip(highs, lows, strict=True):
        if (high >= stop) if sell else (low <= stop):
            return None
        if pos.tp is not None and ((low <= pos.tp) if sell else (high >= pos.tp)):
            return None
        ahead = pos.entry - low if sell else high - pos.entry
        if ahead >= 0.5 * risk:
            stop = pos.entry
            moved = True
    if not moved:
        return None
    return PositionAdvice(new_sl=pos.entry, stop_kind=ExitReason.BREAK_EVEN, note="+0.5 R touched")

            entry_price=price,

            conditions=(
                Condition("h4_flat", True),

    def should_close(self, ctx: StrategyContext, side: Side) -> bool:

    def manage_position(self, ctx: StrategyContext, pos: OpenPosition) -> PositionAdvice | None:

"""Hand-worked H4 lean cases, from close-stamped H1 candles.

The rule measured and frozen on 2026-10-27 (`research/2026-10-26-three-bots/RULES_LOCKED.md`, rule set 2)
is a QUANTITY on the H4 frame, not a D1 label: the close's distance from its own 50-bar mean in its own
14-bar ATR, signed. An earlier version of this module read a D1 regime label; the measurement put D1
last of the three frames (+0.030 against H4's +0.156 correlation with the trade's R).
"""

NEEDED = MEAN_BARS + ATR_BARS  # 64 closed H4 bars


def h1_frame(h4_closes: list[float], *, spread: float = 1.0, forming: float | None = None):
    """H1 candles, CLOSE-stamped as the app's frames are, that resample into the given H4 closes.

    Each H4 bar is four H1 bars: the first three sit at the previous close so the H4 close is the value
    asked for, and every bar spans ``spread`` so the ATR is predictable.
    """
    closes: list[float] = []
    for i, value in enumerate(h4_closes):
        previous = h4_closes[i - 1] if i else value
        closes += [previous, previous, previous, value]
    if forming is not None:
        closes += [forming]  # one extra H1 bar: the H4 bar it belongs to has not closed
    # the first H1 bar CLOSES at 01:00, so it opens at 00:00 and its H4 bar opens at 00:00 too
    index = pd.date_range(datetime(2026, 1, 1, 1, tzinfo=UTC), periods=len(closes), freq="1h")
    frame = pd.DataFrame(
        {
            "open": closes,
            "high": [c + spread / 2 for c in closes],
            "low": [c - spread / 2 for c in closes],
            "close": closes,
        },
        index=index,
    )
    return {Timeframe.H1: frame}, pd.Timestamp(frame.index[-1])


def expected_lean(frames, at) -> float:
    """The same number worked out from the resampled frame, independently of the function under test."""
    bars = resample_from_h1(frames[Timeframe.H1], 4)
    closed = bars[(bars.index + pd.Timedelta(hours=4)) <= at]
    mean = closed.close.rolling(MEAN_BARS).mean().iloc[-1]
    width = atr(closed.high, closed.low, closed.close, ATR_BARS).iloc[-1]
    return float((closed.close.iloc[-1] - mean) / width)


class TestTheLean:
    def test_a_rising_frame_leans_up_and_a_falling_one_leans_down(self):
        up, at_up = h1_frame([100.0 + i for i in range(NEEDED)])
        down, at_down = h1_frame([100.0 - i for i in range(NEEDED)])
        assert edge_lean(up, at_up, Timeframe.M5) > 0
        assert edge_lean(down, at_down, Timeframe.M5) < 0

    def test_a_flat_frame_leans_neither_way(self):
        frames, at = h1_frame([100.0] * NEEDED)
        assert edge_lean(frames, at, Timeframe.M5) == pytest.approx(0.0)
        assert edge_side(frames, at, Timeframe.M5) is None  # exactly zero is on neither side

    def test_the_number_is_the_distance_from_the_mean_in_atrs(self):
        frames, at = h1_frame([100.0 + i for i in range(NEEDED)])
        assert edge_lean(frames, at, Timeframe.M5) == pytest.approx(expected_lean(frames, at))

    def test_a_bar_that_has_not_closed_is_not_read(self):
        # the forming H4 bar carries a huge close; reading it would swamp a flat 64-bar history
        flat, at = h1_frame([100.0] * NEEDED)
        with_forming, at_forming = h1_frame([100.0] * NEEDED, forming=10_000.0)
        assert edge_lean(with_forming, at_forming, Timeframe.M5) == pytest.approx(
            edge_lean(flat, at, Timeframe.M5)
        )

    def test_too_little_history_is_no_answer_rather_than_a_guess(self):
        frames, at = h1_frame([100.0 + i for i in range(NEEDED - 1)])
        assert edge_lean(frames, at, Timeframe.M5) is None
        assert edge_side(frames, at, Timeframe.M5) is None

    def test_a_missing_h1_frame_is_no_answer(self):
        assert edge_lean({}, pd.Timestamp("2026-01-01", tz="UTC"), Timeframe.M5) is None

def true_range(high: pd.Series, low: pd.Series, close: pd.Series) -> pd.Series:
    """``max(H - L, |H - C[t-1]|, |L - C[t-1]|)``; position 0 has no previous close and is NaN."""
    check_aligned(high, low, close)
    return to_series(true_range_values(as_float(high), as_float(low), as_float(close)), close, "true_range")


def atr(high: pd.Series, low: pd.Series, close: pd.Series, n: int = 14) -> pd.Series:
    """Wilder-smoothed true range, in price units; first defined at position *n*."""
    check_period("n", n)
    check_aligned(high, low, close)
    tr = true_range_values(as_float(high), as_float(low), as_float(close))
    return to_series(wilder_values(tr, n), close, f"atr_{n}")


def bollinger(close: pd.Series, n: int = 20, k: float = 2.0) -> pd.DataFrame:
    """Columns ``mid`` (SMA), ``upper``/``lower`` (mid Â± kÂ·Ïƒ, population Ïƒ), ``width`` and ``percent_b``.

    ``width = (upper - lower) / mid``; ``percent_b = (C - lower) / (upper - lower)``, NaN for a flat window.
    All columns are first defined at ``n - 1``.
    """
    check_period("n", n, minimum=2)
    if not (math.isfinite(k) and k > 0):
        raise ConfigError(f"Bollinger k must be > 0 (got {k!r})")
    c = as_float(close)
    mid = rolling_mean(c, n)
    sigma = np.full(c.shape, np.nan, dtype=np.float64)
    flat = np.zeros(c.shape, dtype=bool)
    if len(c) >= n:
        windows = sliding_window_view(c, n)
        sigma[n - 1 :] = windows.std(axis=1, ddof=0)
        # exact flatness test: rolling float sums leave rounding noise where Ïƒ should be 0
        flat[n - 1 :] = np.ptp(windows, axis=1) == 0
    sigma[flat] = 0.0
    upper, lower = mid + k * sigma, mid - k * sigma
    band = upper - lower
    with np.errstate(divide="ignore", invalid="ignore"):
        width = np.where(mid > 0, band / mid, np.nan)
        percent_b = np.where(band > 0, (c - lower) / band, np.nan)
    return pd.DataFrame(
        {"mid": mid, "upper": upper, "lower": lower, "width": width, "percent_b": percent_b},
        index=close.index,
    )


def historical_volatility(close: pd.Series, n: int = 20, *, bars_per_year: float) -> pd.Series:
    """Annualized volatility: sample stdev (ddof=1) of the last *n* log returns Ã— âˆšbars_per_year.

    *bars_per_year* depends on the timeframe and the market's trading hours (``Timeframe.bars_per_year``
    assumes a 24x5 market; crypto trades 24x7). A decimal fraction (0.12 = 12 %); first defined at
    position *n*. Non-positive prices are NaN.
    """
    check_period("n", n, minimum=2)
    if not (math.isfinite(bars_per_year) and bars_per_year > 0):
        raise ConfigError(f"bars_per_year must be > 0 (got {bars_per_year!r})")
    c = as_float(close)
    with np.errstate(divide="ignore", invalid="ignore"):
        log_c = np.where(c > 0, np.log(c), np.nan)
    returns = np.diff(log_c, prepend=np.nan)
    std = pd.Series(returns).rolling(n, min_periods=n).std(ddof=1).to_numpy(dtype=np.float64)
    return to_series(std * math.sqrt(bars_per_year), close, f"hv_{n}")


def atr_percentile(atr_values: pd.Series, lookback: int = 100) -> pd.Series:
    """Where the current ATR sits among the previous ``lookback - 1`` values, in [0, 100].

    ``100 Â· count(prev â‰¤ current) / (lookback - 1)`` over the window ending at *t*: 100 means the highest
    ATR of the window, 0 the lowest. Defined once *lookback* consecutive finite ATR values exist
    (for ATR(n): at position ``n + lookback - 1``).
    """
    check_period("lookback", lookback, minimum=2)
    a = as_float(atr_values)
    out = np.full(a.shape, np.nan, dtype=np.float64)
    if len(a) >= lookback:
        windows = sliding_window_view(a, lookback)
        current = windows[:, -1]
        counts = (windows[:, :-1] <= current[:, None]).sum(axis=1)
        pct = 100.0 * counts / (lookback - 1)
        complete = np.isfinite(windows).all(axis=1)
        out[lookback - 1 :] = np.where(complete, pct, np.nan)
    return to_series(out, atr_values, f"atr_pct_{lookback}")

def h1_frame(h4_closes: list[float], *, spread: float = 1.0, forming: float | None = None):
    """H1 candles, CLOSE-stamped as the app's frames are, that resample into the given H4 closes.

    Each H4 bar is four H1 bars: the first three sit at the previous close so the H4 close is the value
    asked for, and every bar spans ``spread`` so the ATR is predictable.
    """
    closes: list[float] = []
    for i, value in enumerate(h4_closes):
        previous = h4_closes[i - 1] if i else value
        closes += [previous, previous, previous, value]
    if forming is not None:
        closes += [forming]  # one extra H1 bar: the H4 bar it belongs to has not closed
    # the first H1 bar CLOSES at 01:00, so it opens at 00:00 and its H4 bar opens at 00:00 too
    index = pd.date_range(datetime(2026, 1, 1, 1, tzinfo=UTC), periods=len(closes), freq="1h")
    frame = pd.DataFrame(
        {
            "open": closes,
            "high": [c + spread / 2 for c in closes],
            "low": [c - spread / 2 for c in closes],
            "close": closes,
        },
        index=index,
    )
    return {Timeframe.H1: frame}, pd.Timestamp(frame.index[-1])


def expected_lean(frames, at) -> float:
    """The same number worked out from the resampled frame, independently of the function under test."""
    bars = resample_from_h1(frames[Timeframe.H1], 4)
    closed = bars[(bars.index + pd.Timedelta(hours=4)) <= at]
    mean = closed.close.rolling(MEAN_BARS).mean().iloc[-1]
    width = atr(closed.high, closed.low, closed.close, ATR_BARS).iloc[-1]
    return float((closed.close.iloc[-1] - mean) / width)


class TestTheLean:
    def test_a_rising_frame_leans_up_and_a_falling_one_leans_down(self):
        up, at_up = h1_frame([100.0 + i for i in range(NEEDED)])
        down, at_down = h1_frame([100.0 - i for i in range(NEEDED)])
        assert edge_lean(up, at_up, Timeframe.M5) > 0
        assert edge_lean(down, at_down, Timeframe.M5) < 0

    def test_a_flat_frame_leans_neither_way(self):
        frames, at = h1_frame([100.0] * NEEDED)
        assert edge_lean(frames, at, Timeframe.M5) == pytest.approx(0.0)
        assert edge_side(frames, at, Timeframe.M5) is None  # exactly zero is on neither side

    def test_the_number_is_the_distance_from_the_mean_in_atrs(self):
        frames, at = h1_frame([100.0 + i for i in range(NEEDED)])
        assert edge_lean(frames, at, Timeframe.M5) == pytest.approx(expected_lean(frames, at))

    def test_a_bar_that_has_not_closed_is_not_read(self):
        # the forming H4 bar carries a huge close; reading it would swamp a flat 64-bar history
        flat, at = h1_frame([100.0] * NEEDED)
        with_forming, at_forming = h1_frame([100.0] * NEEDED, forming=10_000.0)
        assert edge_lean(with_forming, at_forming, Timeframe.M5) == pytest.approx(
            edge_lean(flat, at, Timeframe.M5)
        )

    def test_too_little_history_is_no_answer_rather_than_a_guess(self):
        frames, at = h1_frame([100.0 + i for i in range(NEEDED - 1)])
        assert edge_lean(frames, at, Timeframe.M5) is None
        assert edge_side(frames, at, Timeframe.M5) is None

    def test_a_missing_h1_frame_is_no_answer(self):
        assert edge_lean({}, pd.Timestamp("2026-01-01", tz="UTC"), Timeframe.M5) is None

def bollinger(close: pd.Series, n: int = 20, k: float = 2.0) -> pd.DataFrame:
    """Columns ``mid`` (SMA), ``upper``/``lower`` (mid Â± kÂ·Ïƒ, population Ïƒ), ``width`` and ``percent_b``.

    ``width = (upper - lower) / mid``; ``percent_b = (C - lower) / (upper - lower)``, NaN for a flat window.
    All columns are first defined at ``n - 1``.
    """
    check_period("n", n, minimum=2)
    if not (math.isfinite(k) and k > 0):
        raise ConfigError(f"Bollinger k must be > 0 (got {k!r})")
    c = as_float(close)
    mid = rolling_mean(c, n)
    sigma = np.full(c.shape, np.nan, dtype=np.float64)
    flat = np.zeros(c.shape, dtype=bool)
    if len(c) >= n:
        windows = sliding_window_view(c, n)
        sigma[n - 1 :] = windows.std(axis=1, ddof=0)
        # exact flatness test: rolling float sums leave rounding noise where Ïƒ should be 0
        flat[n - 1 :] = np.ptp(windows, axis=1) == 0
    sigma[flat] = 0.0
    upper, lower = mid + k * sigma, mid - k * sigma
    band = upper - lower
    with np.errstate(divide="ignore", invalid="ignore"):
        width = np.where(mid > 0, band / mid, np.nan)
        percent_b = np.where(band > 0, (c - lower) / band, np.nan)
    return pd.DataFrame(
        {"mid": mid, "upper": upper, "lower": lower, "width": width, "percent_b": percent_b},
        index=close.index,
    )


def historical_volatility(close: pd.Series, n: int = 20, *, bars_per_year: float) -> pd.Series:
    """Annualized volatility: sample stdev (ddof=1) of the last *n* log returns Ã— âˆšbars_per_year.

    *bars_per_year* depends on the timeframe and the market's trading hours (``Timeframe.bars_per_year``
    assumes a 24x5 market; crypto trades 24x7). A decimal fraction (0.12 = 12 %); first defined at
    position *n*. Non-positive prices are NaN.
    """
    check_period("n", n, minimum=2)
    if not (math.isfinite(bars_per_year) and bars_per_year > 0):
        raise ConfigError(f"bars_per_year must be > 0 (got {bars_per_year!r})")
    c = as_float(close)
    with np.errstate(divide="ignore", invalid="ignore"):
        log_c = np.where(c > 0, np.log(c), np.nan)
    returns = np.diff(log_c, prepend=np.nan)
    std = pd.Series(returns).rolling(n, min_periods=n).std(ddof=1).to_numpy(dtype=np.float64)
    return to_series(std * math.sqrt(bars_per_year), close, f"hv_{n}")


def atr_percentile(atr_values: pd.Series, lookback: int = 100) -> pd.Series:
    """Where the current ATR sits among the previous ``lookback - 1`` values, in [0, 100].

    ``100 Â· count(prev â‰¤ current) / (lookback - 1)`` over the window ending at *t*: 100 means the highest
    ATR of the window, 0 the lowest. Defined once *lookback* consecutive finite ATR values exist
    (for ATR(n): at position ``n + lookback - 1``).
    """
    check_period("lookback", lookback, minimum=2)
    a = as_float(atr_values)
    out = np.full(a.shape, np.nan, dtype=np.float64)
    if len(a) >= lookback:
        windows = sliding_window_view(a, lookback)
        current = windows[:, -1]
        counts = (windows[:, :-1] <= current[:, None]).sum(axis=1)
        pct = 100.0 * counts / (lookback - 1)
        complete = np.isfinite(windows).all(axis=1)
        out[lookback - 1 :] = np.where(complete, pct, np.nan)
    return to_series(out, atr_values, f"atr_pct_{lookback}")

def condition_strength(conditions: Iterable[Condition]) -> float:
    """Weighted share of passed conditions, 0â€“100 (Â§A26 setup strength before confluence enrichment)."""
    items = list(conditions)
    total = sum(c.weight for c in items)
    if total <= 0:
        return 0.0
    return 100.0 * sum(c.weight for c in items if c.passed) / total

# Review: H4 edge-side gate

**Scope:** Read the brief and reviewed `edge_side.py`, both strategy call paths, ATR implementation, the 11 unit tests, registry error handling, and lane warmup configuration. No files were changed. The report is printed here as requested.

## 1. Open/close convention

**Checked:** `resample_from_h1` in [edge_side.py](C:/Users/korap/taa/app/strategy/edge_side.py:33) and the app’s close-stamped frame contract in `StrategyContext`.

**How:** App H1 candles are stamped at close. Shifting each timestamp back one hour assigns the candle to its opening hour. For a four-hour bucket opening at 00:00, the H1 candles closing at 01:00 through 04:00 are assigned to opens 00:00 through 03:00. The filter `(bucket open + 4 hours) <= at` includes that bar only at or after 04:00. It excludes the forming H4 bar.

**Verdict:** The convention and closure boundary are correct for UTC close-stamped H1 frames. I found no one-hour lookahead in this path.

## 2. Fail-closed behavior

**Checked:** Missing and short input handling in `edge_lean`, and the `behind` gate conditions in both strategies.

**How:** Missing H1 or missing `close` returns `None`; fewer than 64 closed resampled bars returns `None`; zero or NaN ATR width fails `not width > 0`; a NaN rolling mean is rejected. `behind(None, side)` remains `None`. Both strategies hold when the result is `None` or negative. The trend strategy evaluates the edge gate before reading the H1 frame for its setup; the range strategy reads its required frames first, but checks the gate before producing an entry.

**Finding:** The returned lean is not checked for finiteness. If malformed input contains infinities, the calculation can produce a NaN lean: the mean and width checks do not reject infinity, and `(inf - inf) / inf` is NaN. Then `lean < 0` is false, so both gates can pass. This requires non-finite candle data, but it is a fail-open path if such data reaches these functions. A bad frame shape or resampling error can also raise; the strategy registry catches strategy exceptions and returns `STRATEGY_ERROR` HOLD, so those exceptions do not escape the engine loop.

**Verdict:** Missing, short, and zero-width cases fail closed. Non-finite results do not reliably fail closed. The direct risk is allowing an entry when the edge value is invalid, if corrupted non-finite H1 data reaches a strategy.

## 3. ATR ruler and warmup

**Checked:** `app.indicators.volatility.atr`, its Wilder seeding, edge-side bar requirement, and configured warmup.

**How:** ATR uses true range and Wilder smoothing, seeded with the first full 14-value window. The first true range is NaN because there is no prior close; the first ATR is therefore defined at position 14. At 64 closed H4 bars, the final 50-bar mean is defined and ATR is defined. The H4 bars are resampled from H1, so 64 H4 bars require up to 256 H1 bars; the configured 1,200 H1 bars exceed that requirement.

**Verdict:** The app ATR ruler is being used, and 64 bars suffice for a non-NaN result. The `three_bots` lane has `warmup_bars: 1200`; trend strategies also declare 1,200 when this gate is enabled, and `range_break` declares 1,200. I found no other configured lane enabling `require_edge_side`.
